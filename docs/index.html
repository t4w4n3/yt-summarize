04Pipeline sequence

%% Summarize YT — pipeline sequence
sequenceDiagram
  autonumber
  participant U as Browser
  participant A as App :8080
  participant DB as SQLite jobs.db
  participant W as Worker
  participant Y as yt-dlp
  participant F as ffmpeg
  participant O as OpenRouter

  U->>A: POST /api/summarize {url}
  A->>A: validate (host · 11-char video id)
  A->>DB: INSERT job → queued
  A-->>U: 201 {jobId}

  loop poll every 2 s
    U->>A: GET /api/jobs/:id
    A->>DB: SELECT job
    A-->>U: {status, stage, progress}
  end

  W->>DB: claimNextJob() → running
  W->>Y: yt-dlp (via Mullvad SOCKS5) → audio file
  W->>F: ffmpeg → 16 kHz mono WAV
  W->>O: POST /audio/transcriptions → transcript.txt
  W->>O: POST /chat/completions (system prompt + transcript) → Markdown
  W->>DB: markDone(title, markdown)
  W->>DB: summary.md → artifacts volume

  U->>A: GET /api/jobs/:id/result
  A-->>U: {title, markdown, wordCount}
  U->>A: GET /api/jobs/:id/result.md
  A-->>U: text/markdown attachment
